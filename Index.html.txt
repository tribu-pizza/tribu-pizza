  document.querySelectorAll('a[href^="#"]').forEach(anchor => {
    anchor.addEventListener('click', function (e) {

      const target = document.querySelector(this.getAttribute('href'));
      if (target) { window.scrollTo({ top: target.offsetTop - 70, behavior: 'smooth' }); }
    });

  async function sendOrder() {
    const btn = document.querySelector('.submit-btn');
    const name = document.getElementById('custName').value;
    const phone = document.getElementById('custPhone').value;
    const email = document.getElementById('custEmail').value;
    const eventType = document.getElementById('custEventType').value;
    const town = document.getElementById('custTown').value;
    const date = document.getElementById('custDate').value;
    const hours = document.getElementById('custHours').value;
    const qty = document.getElementById('custPizzaQty').value;
    const notes = document.getElementById('custNotes').value;
    const pizzas = Array.from(document.querySelectorAll('input[type="checkbox"]:checked')).map(cb => cb.value).join(', ');

    if (!name || !phone || !town || !email || !date) {
      alert("Por favor completa todos los campos obligatorios (*).");
      return;
    }

    btn.disabled = true;
    btn.innerText = "Enviando...";

    const formData = {
      Nombre: name,
      Tel: phone,
      Email: email,
      Evento: eventType,
      Pueblo: town,
      Fecha: date,
      Horas: hours,
      Cantidad: qty,
      Sabores: pizzas,
      Notas: notes
    };

    try {
      await fetch('https://formspree.io/f/xaqvojwn', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' },
        body: JSON.stringify(formData)
      });
      
      const text = `*SOLICITUD DE RESERVA TRIBU PIZZA*\n\n*Cliente:* ${name}\n*Email:* ${email}\n*Tel:* ${phone}\n*Evento:* ${eventType}\n*Pueblo:* ${town}\n*Fecha/Hora:* ${date}\n*Horas Servicio:* ${hours}\n*Cantidad:* ${qty}\n*Sabores:* ${pizzas}\n*Notas:* ${notes || 'Ninguna'}`;
      
      window.open(`https://wa.me/17874526292?text=${encodeURIComponent(text)}`, '_blank');

      document.getElementById('bookingFormContainer').style.display = 'none';
      document.getElementById('thanksMsg').style.display = 'block';
      window.scrollTo({ top: document.getElementById('booking').offsetTop - 70, behavior: 'smooth' });
      
    } catch (e) {
      alert("Hubo un error al enviar tu solicitud.");
      btn.disabled = false;
      btn.innerText = "🔥 Enviar Solicitud de Reserva";
    }
  }
</script>
